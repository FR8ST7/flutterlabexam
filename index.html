SET 11
LayoutBuilder Responsive Expanded Row / Column

Responsive Restaurant Menu

Uses LayoutBuilder to switch between a vertical Column on mobile (<600px) and a horizontal Row with Expanded on larger screens.

lib/main.dart • 25 lines 
import 'package:flutter/material.dart';

void main() => runApp(MaterialApp(home: Scaffold(
  appBar: AppBar(title: Text("Restaurant Menu")),
  body: LayoutBuilder(builder: (context, c) {
    var burger = Card(child: ListTile(
      title: Text("Burger"), subtitle: Text("Rs.120")));
    var pizza = Card(child: ListTile(
      title: Text("Pizza"), subtitle: Text("Rs.200")));
    var sandwich = Card(child: ListTile(
      title: Text("Sandwich"), subtitle: Text("Rs.80")));
    if (c.maxWidth < 600) {
      return Column(children: [burger, pizza, sandwich]);
    }
    return Row(children: [
      Expanded(child: burger), Expanded(child: pizza),
      Expanded(child: sandwich),
    ]);
  }),
)));

Form GlobalKey<FormState> TextFormField SnackBar

Library Membership Registration

Interactive form handling user inputs (Name, Register No, Email) with field validation and feedback via ScaffoldMessenger SnackBar.

lib/main.dart • 28 lines 
import 'package:flutter/material.dart';

void main() => runApp(MaterialApp(home: Home()));

class Home extends StatelessWidget {
  final formKey = GlobalKey();

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: Text("Library Registration")),
    body: Padding(padding: EdgeInsets.all(15),
      child: Form(key: formKey, child: Column(children: [
        TextFormField(decoration: InputDecoration(labelText: "Name"),
          validator: (v) => v!.isEmpty ? "Enter name" : null),
        TextFormField(decoration: InputDecoration(labelText: "Register No"),
          validator: (v) => v!.isEmpty ? "Enter number" : null),
        TextFormField(decoration: InputDecoration(labelText: "Email"),
          validator: (v) => v!.isEmpty ? "Enter email" : null),
        ElevatedButton(child: Text("Register"), onPressed: () {
          if (formKey.currentState!.validate()) {
            ScaffoldMessenger.of(context).showSnackBar(
              SnackBar(content: Text("Membership Registered Successfully")));
          }
        }),
      ])),
    ),
  );
}

Tamil Nadu Tourist Places Gallery

Displays a 2-column image gallery using GridView.count and bundled assets for Ooty, Kodaikanal, and Kanyakumari.

Asset Configuration Steps
  1. Create folder assets/images/ in the root of your project.
  2. Place image files: ooty.jpg, kodaikanal.jpg, kanyakumari.jpg.
  3. Add to pubspec.yaml under flutter:: 
    assets:
  - assets/images/
  4. Run flutter pub get.
lib/main.dart • 25 lines 
import 'package:flutter/material.dart';

void main() => runApp(MaterialApp(home: Scaffold(
  appBar: AppBar(title: Text("Tamil Nadu Tourism Gallery")),
  body: GridView.count(
    crossAxisCount: 2, crossAxisSpacing: 10, mainAxisSpacing: 10,
    children: [
      place("Ooty", "assets/images/ooty.jpg"),
      place("Kodaikanal", "assets/images/kodaikanal.jpg"),
      place("Kanyakumari", "assets/images/kanyakumari.jpg"),
    ],
  ),
)));

Widget place(String name, String image) => Card(child: Column(
  children: [
    Expanded(child: Image.asset(image, width: double.infinity,
      fit: BoxFit.cover)),
    Text(name),
  ],
));

Traffic Signal Animation

Demonstrates Flutter's implicit animations using AnimatedContainer that animates size and color transitions between STOP (Red) and GO (Green).

lib/main.dart • 32 lines 
import 'package:flutter/material.dart';

void main() => runApp(MaterialApp(home: Home()));

class Home extends StatefulWidget {
  @override
  State createState() => _HomeState();
}

class _HomeState extends State {
  bool green = false;

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: Text("Traffic Signal Animation")),
    body: Center(child: Column(
      mainAxisAlignment: MainAxisAlignment.center,
      children: [
        AnimatedContainer(
          duration: Duration(seconds: 1),
          width: green ? 200 : 150, height: green ? 200 : 150,
          color: green ? Colors.green : Colors.red),
        Text(green ? "GO" : "STOP"),
        ElevatedButton(child: Text("Change Signal"), onPressed: () {
          setState(() => green = !green);
        }),
      ],
    )),
  );
}
SET 16
JSON Model toJson / fromJson ListView.builder State Management

Employee Record Manager

Implements a custom Dart data model class with JSON serialization (toJson() and fromJson()) and updates dynamic records in a ListView.builder.

lib/main.dart • 41 lines 
import 'package:flutter/material.dart';

class Employee {
  int id;
  String name, department;
  Employee(this.id, this.name, this.department);

  Map toJson() =>
      {"id": id, "name": name, "department": department};

  factory Employee.fromJson(Map json) =>
      Employee(json["id"], json["name"], json["department"]);
}

void main() => runApp(MaterialApp(home: Home()));

class Home extends StatefulWidget {
  @override
  State createState() => _HomeState();
}

class _HomeState extends State {
  Employee emp = Employee(201, "Ravi", "IT");
  List names = [];

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: Text("Employee Record Manager")),
    body: Column(children: [
      Text("ID: 201  Name: Ravi  Department: IT"),
      ElevatedButton(child: Text("Add Employee"), onPressed: () {
        setState(() => names.add(emp.name));
      }),
      Text("JSON: ${emp.toJson()}"),
      Expanded(child: ListView.builder(
        itemCount: names.length,
        itemBuilder: (context, i) => ListTile(title: Text(names[i])))),
    ]),
  );
}
