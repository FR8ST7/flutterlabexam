SET 12 – LIBRARY MEMBERSHIP REGISTRATION
============================================================
import 'package:flutter/material.dart';

void main() => runApp(MaterialApp(home: Home()));

class Home extends StatelessWidget {
  final formKey = GlobalKey<FormState>();

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: Text("Library Registration")),
    body: Padding(padding: EdgeInsets.all(15),
      child: Form(key: formKey, child: Column(children: [
        TextFormField(decoration: InputDecoration(labelText: "Name"),
          validator: (v) => v!.isEmpty ? "Enter name" : null),
        TextFormField(decoration: InputDecoration(labelText: "Register No"),
          validator: (v) => v!.isEmpty ? "Enter number" : null),
        TextFormField(decoration: InputDecoration(labelText: "Email"),
          validator: (v) => v!.isEmpty ? "Enter email" : null),
        ElevatedButton(child: Text("Register"), onPressed: () {
          if (formKey.currentState!.validate()) {
            ScaffoldMessenger.of(context).showSnackBar(
              SnackBar(content: Text("Membership Registered Successfully")));
          }
        }),
      ])),
    ),
  );
}

SET 13 – TAMIL NADU TOURIST PLACES GALLERY
============================================================
import 'package:flutter/material.dart';

void main() => runApp(MaterialApp(home: Scaffold(
  appBar: AppBar(title: Text("Tamil Nadu Tourism Gallery")),
  body: GridView.count(
    crossAxisCount: 2, crossAxisSpacing: 10, mainAxisSpacing: 10,
    children: [
      place("Ooty", "assets/images/ooty.jpg"),
      place("Kodaikanal", "assets/images/kodaikanal.jpg"),
      place("Kanyakumari", "assets/images/kanyakumari.jpg"),
    ],
  ),
))));

Widget place(String name, String image) => Card(child: Column(
  children: [
    Expanded(child: Image.asset(image, width: double.infinity,
      fit: BoxFit.cover)),
    Text(name),
  ],
));

ASSET SETUP:
1. Create assets/images inside your project.
2. Add ooty.jpg, kodaikanal.jpg and kanyakumari.jpg.
3. Add these lines under flutter: in pubspec.yaml:
   assets:
     - assets/images/
4. Run: flutter pub get

SET 15 – TRAFFIC SIGNAL ANIMATION
============================================================
import 'package:flutter/material.dart';

void main() => runApp(MaterialApp(home: Home()));

class Home extends StatefulWidget {
  @override
  State<Home> createState() => _HomeState();
}

class _HomeState extends State<Home> {
  bool green = false;

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: Text("Traffic Signal Animation")),
    body: Center(child: Column(
      mainAxisAlignment: MainAxisAlignment.center,
      children: [
        AnimatedContainer(
          duration: Duration(seconds: 1),
          width: green ? 200 : 150, height: green ? 200 : 150,
          color: green ? Colors.green : Colors.red),
        Text(green ? "GO" : "STOP"),
        ElevatedButton(child: Text("Change Signal"), onPressed: () {
          setState(() => green = !green);
        }),
      ],
    )),
  );
}

SET 16 – EMPLOYEE RECORD MANAGER
============================================================
import 'package:flutter/material.dart';

class Employee {
  int id;
  String name, department;
  Employee(this.id, this.name, this.department);

  Map<String, dynamic> toJson() =>
      {"id": id, "name": name, "department": department};

  factory Employee.fromJson(Map<String, dynamic> json) =>
      Employee(json["id"], json["name"], json["department"]);
}

void main() => runApp(MaterialApp(home: Home()));

class Home extends StatefulWidget {
  @override
  State<Home> createState() => _HomeState();
}

class _HomeState extends State<Home> {
  Employee emp = Employee(201, "Ravi", "IT");
  List<String> names = [];

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: Text("Employee Record Manager")),
    body: Column(children: [
      Text("ID: 201  Name: Ravi  Department: IT"),
      ElevatedButton(child: Text("Add Employee"), onPressed: () {
        setState(() => names.add(emp.name));
      }),
      Text("JSON: ${emp.toJson()}"),
      Expanded(child: ListView.builder(
        itemCount: names.length,
        itemBuilder: (context, i) => ListTile(title: Text(names[i])))),
    ]),
  );
}
